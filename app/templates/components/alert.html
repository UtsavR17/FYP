{# components/alert.html Reusable inline alert macros for page-level status messages. These are distinct from flash messages (which are post-redirect feedback). Use these for static content-area messages such as search result banners. Usage: {% from 'components/alert.html' import info_alert, warning_alert %} {{ info_alert('Showing results for: ' ~ search_query) }} {{ warning_alert('No records match your search. Try a different keyword.') }} #} {# ----------------------------------------------------------------------- info_alert Blue informational alert. Use for search result banners, tips, or neutral status messages. ----------------------------------------------------------------------- #} {% macro info_alert(message, dismissible=False) %}
{{ message }} {% if dismissible %} {% endif %}
{% endmacro %} {# ----------------------------------------------------------------------- warning_alert Yellow warning alert. Use for soft warnings such as no-search-results messages or data integrity notices. ----------------------------------------------------------------------- #} {% macro warning_alert(message, dismissible=False) %}
{{ message }} {% if dismissible %} {% endif %}
{% endmacro %}
